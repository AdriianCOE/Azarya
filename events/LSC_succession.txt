# Step 2 — Viktor dies. Retires him and promotes Kael in the same effect
# (mirrors vanilla Spain: spain.1 immediate retires the old SPR leader and

# Czardom is never left with no leader between the two events.

	id = LSC_succession.12
	title = LSC_succession.12.t
	desc = LSC_succession.12.d

		has_country_flag = LSC_route_kael
	}

	is_triggered_only = yes
	major = yes

	immediate = {
		hidden_effect = {
			if = {
				limit = {
					has_country_leader = {
						character = CZL_viktor_iv_vorenkov

					}

# Step 6 — The Marble Gambit resolves. Alvaro's posture is decided first and
# independently (he is never a third belligerent here -- see events/
# LSC_civil_war.txt's shared resolution system for how his flags are read
# later, and Part D's proposal for how they pay off). The succession itself
# then resolves to exactly one of three endings via a priority if/else_if
# chain over the accumulated variables, worst-case (limited civil war) last.
country_event = {
	id = LSC_succession.46
	title = LSC_succession.46.t
	desc = LSC_succession.46.d
	picture = GFX_news_event_generic_parliament

	trigger = {
		has_country_flag = LSC_route_selene
		has_country_flag = LSC_selene_holds_regicide_evidence
		NOT = { has_global_flag = LSC_succession_war_active_global }
		NOT = { has_global_flag = LSC_succession_war_resolved_global }
	}

	is_triggered_only = yes
	major = yes

	immediate = {
		hidden_effect = {
			# Alvaro's posture: an independent, single-flag read of the same
			# variables, not a war of his own (section 8 of the brief).
			if = {
				limit = { check_variable = { LSC_revolutionary_strength > 55 } }
				set_country_flag = LSC_alvaro_demands_autonomy
			}
			else_if = {
				limit = { check_variable = { LSC_northern_unrest > 55 } }
				set_country_flag = LSC_alvaro_armed_neutrality
			}
			else = {
				set_country_flag = LSC_alvaro_supports_selene
			}

			# The three-way succession outcome itself.
			if = {
				limit = {
					check_variable = { LSC_selene_legitimacy > 55 }
					check_variable = { LSC_army_loyalty < 55 }
					check_variable = { LSC_oligarch_influence < 55 }
					check_variable = { LSC_northern_unrest < 55 }
				}
				# RESULT 1 — The Marble Settlement (no start_civil_war). Kael
				# is recruited (so future content can use him) but never
				# retired -- Part D's own audit found no vanilla proof that
				# retire_character leaves a character safely reusable, so his
				# disposition is tracked purely by flag instead, per the
				# user's explicit correction: keep the token accessible for
				# future interrogation/trial/exile content.
				remove_ideas = { LSC_gathering_storm }
				recruit_character = CZL_kael_vorenkov
				set_country_flag = LSC_kael_under_surveillance
				add_stability = 0.03
				add_ideas = { LSC_selene_marble_regency }
				LSC_clear_succession_results = yes
				set_country_flag = LSC_selene_victory
				set_country_flag = LSC_selene_peaceful_settlement
				set_global_flag = LSC_succession_result_locked_global
				set_global_flag = LSC_succession_war_resolved_global
				country_event = { id = LSC_succession.47 hours = 6 }
			}
			else_if = {
				limit = {
					check_variable = { LSC_selene_legitimacy > 45 }
					NOT = { check_variable = { LSC_northern_unrest > 65 } }
					OR = {
						check_variable = { LSC_army_loyalty > 55 }
						check_variable = { LSC_oligarch_influence > 55 }
					}
				}
				# RESULT 2 — Palace Coup (no start_civil_war, no territory
				# split). Kael was never CZL's ruling leader on this route, so
				# the defensive retire_country_leader check from the original
				# draft is dropped as dead code; same reasoning as Result 1
				# applies to keeping his token intact rather than retiring it.
				recruit_character = CZL_kael_vorenkov
				set_country_flag = LSC_kael_imprisoned
				add_stability = -0.08
				add_war_support = -0.05
				add_ideas = { LSC_selene_court_under_siege }
				add_to_variable = { var = LSC_northern_unrest value = 10 }
				clamp_variable = { var = LSC_northern_unrest min = 0 max = 100 }
				LSC_clear_succession_results = yes
				set_country_flag = LSC_selene_victory
				set_country_flag = LSC_selene_palace_coup
				set_global_flag = LSC_succession_result_locked_global
				set_global_flag = LSC_succession_war_resolved_global
				country_event = { id = LSC_succession.48 hours = 6 }
			}
			else = {
				# RESULT 3 — Limited civil war. Only this branch uses
				# start_civil_war. Kael's breakaway faction gets the three
				# contiguous southern states audited for this purpose (329
				# Ponta, 317 Caralis, 304 Castellum -- real land adjacency
				# confirmed via map/provinces.bmp; 317 holds the only other
				# supply node in the old 8-state south besides Aelia itself).
				# Selene keeps Aelia, the Crown, the Court and the remaining
				# eleven states, including every other state ever considered
				# for the earlier Kael/Alvaro 8-state split.
				recruit_character = CZL_kael_vorenkov
				set_country_flag = LSC_succession_side_selene
				set_global_flag = LSC_succession_war_active_global

				add_ideas = { LSC_selene_divided_guard }

				if = {
					limit = { check_variable = { LSC_army_loyalty > 65 } }
					set_country_flag = LSC_war_bonus_selene_loyalty
				}
				if = {
					limit = { check_variable = { LSC_selene_legitimacy > 65 } }
					add_war_support = 0.03
				}

				start_civil_war = {
					ruling_party = neutrality
					ideology = fascism
					size = 0.25
					capital = 317
					states = {
						329
						317
						304
					}
					PREV = { # Original country (CZL, still ROOT throughout this effect)
						CZL_kael_vorenkov = {
							set_nationality = PREV.PREV
						}
					}
				}

				random_other_country = {
					limit = {
						original_tag = CZL
						NOT = { tag = CZL }
					}
					set_cosmetic_tag = LSC_OLD_REGIME
					set_country_flag = LSC_succession_side_old_regime
					set_stability = 0.35
					set_war_support = 0.60
					add_ideas = { LSC_kael_emergency_government }
					add_country_leader_role = {
						character = CZL_kael_vorenkov
						country_leader = {
							ideology = fascism_ideology
							traits = { }
							expire = "1965.1.1"
						}
						promote_leader = yes
					}
					if = {
						limit = { ROOT = { has_country_flag = LSC_war_bonus_selene_loyalty } }
						add_war_support = 0.03
					}
				}
			}
		}
	}

	option = {
		name = LSC_succession.46.a
	}
}
