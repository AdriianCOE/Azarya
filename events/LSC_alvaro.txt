				has_country_flag = LSC_selene_holds_regicide_evidence
			}
		}
		add_political_power = 20
		add_stability = -0.01
		set_country_flag = LSC_alvaro_truth_commissioned
		set_country_flag = LSC_alvaro_viktor_inquiry_open
		ai_chance = { factor = 100 }
	}
	option = {
		name = LSC_alvaro.2.c
		trigger = {
			NOT = {
				OR = {
					has_country_flag = LSC_kael_regicide_proof_destroyed
					has_country_flag = LSC_kael_regicide_proof_archived_by_crown
					has_country_flag = LSC_kael_regicide_proof_known_to_staff
					has_country_flag = LSC_selene_holds_regicide_evidence
				}
			}
		}
		add_stability = 0.02
		set_country_flag = LSC_alvaro_innocence_declared
		set_country_flag = LSC_alvaro_viktor_inquiry_open
		ai_chance = { factor = 100 }
	}
}
