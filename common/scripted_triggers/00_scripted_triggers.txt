					}
				} 
			}
			FROM = { 
				OR = {
					has_tech = centimetric_radar 
					has_tech = phased_array
				}
			}
		}
		else_if = {
			limit = {
				ROOT = { 
					has_tech = phased_array 
					NOT = {
						has_tech = monopulse_radar
					}
				} 
			}

		else_if = {
			limit = {

						has_tech = future_radar
					}
				}
