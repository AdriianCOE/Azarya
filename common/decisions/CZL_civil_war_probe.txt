	# ============================================================
	# RUPTURE 1/3 — Kael shape. Sets the global probe-active flag that the
	# four on_action hooks now require. available (not visible) is what
	# gates the has_character check — the decision is greyed out, not
	# hidden, until CZL_alvaro_vorenkov exists. Never recruits him here.

		priority = 100
		allowed = { tag = CZL }
		available = {

			NOT = { has_country_flag = LSC_debug_war_active }
			has_character = CZL_alvaro_vorenkov
		}
		fire_only_once = yes
		complete_effect = {
			set_global_flag = LSC_debug_civil_war_probe_active
			log = "[GetDateAsString] LSC_DEBUG TEST1: firing on [ROOT.GetName] ([ROOT.GetTag]). Precondition confirmed: CZL_alvaro_vorenkov exists in this country."

				}
				PREV = {
					# Mexico.txt PREV={CHARACTER={...}} smoke test — deliberately NOT
					# calling set_nationality yet (target expression unverified for
					# our scope chain). Only confirms the block resolves.
					CZL_alvaro_vorenkov = {
						log = "[GetDateAsString] LSC_DEBUG TEST1: PREV block resolved to a character scope for CZL_alvaro_vorenkov"
					}

			}
			log = "[GetDateAsString] LSC_DEBUG TEST1: start_civil_war returned. CZL capital now: [ROOT.GetCapitalState.GetName]. Run 'Identify Rebel' next in this same save."
		}
	}

	# ============================================================
	# RUPTURE 2/3 — Alvaro shape, single fixed order:
	#   log initial capital -> set_capital on CZL (CZL's own scope, no
	#   scope switch happened, ROOT is still CZL throughout this whole
	#   effect) -> log new capital -> start_civil_war (8 south states,
	#   rebel capital 335) -> log final capital -> inline rebel-candidate
	#   log (informational only — does NOT set LSC_debug_dynamic_rebel;
	#   that stays the exclusive job of Identify Rebel, see item 5).
	# ============================================================
	LSC_debug_test_2_south_split = {
		icon = GFX_decision_generic_military
		priority = 90
		allowed = { tag = CZL }
		available = {
			has_country_flag = LSC_debug_civil_war_probe
			NOT = { has_country_flag = LSC_debug_war_active }
		}
		fire_only_once = yes
		complete_effect = {
			set_global_flag = LSC_debug_civil_war_probe_active
			log = "[GetDateAsString] LSC_DEBUG TEST2 step1: CZL capital BEFORE anything: [ROOT.GetCapitalState.GetName]"
			set_capital = { state = 303 }
			log = "[GetDateAsString] LSC_DEBUG TEST2 step2: CZL capital AFTER set_capital={state=303} (still pre-war): [ROOT.GetCapitalState.GetName]"
			set_country_flag = LSC_debug_war_active
			start_civil_war = {
				ruling_party = communism
				ideology = neutrality
				size = 0.5
				capital = 335
				states = {
					335 360 353 329 323 317 305 304
				}
			}
			log = "[GetDateAsString] LSC_DEBUG TEST2 step3: start_civil_war returned. CZL capital now: [ROOT.GetCapitalState.GetName] (should still read 303 — confirm this survives save/load separately)"
			every_other_country = {
				limit = {
					original_tag = CZL
					NOT = { tag = CZL }
				}
				log = "[GetDateAsString] LSC_DEBUG TEST2 step4 (informational only, does not flag): rebel candidate = [THIS.GetName] ([THIS.GetTag]) capital=[THIS.GetCapitalState.GetName]"
			}
		}
	}

	# ============================================================
	# RUPTURE 3/3 — does NOT start a new war. Only available once exactly
	# one country is already marked LSC_debug_dynamic_rebel by Test 1 +

	LSC_debug_test_3_oob_controlled = {

				NOT = { tag = CZL }

		}
		fire_only_once = yes
		complete_effect = {

			}
		}
	}

	# ============================================================
	# IDENTIFY REBEL — clears any stale mark first, zeroes a counter,
	# counts real matches, logs every candidate, and only marks + confirms
	# success when the count is exactly 1. A count of 0 or 2+ logs an
	# explicit error instead of guessing which one to mark.
	# ============================================================
	LSC_debug_identify_rebel = {
		icon = GFX_decision_generic_decision
		priority = 70
		allowed = { tag = CZL }
		available = {
			has_country_flag = LSC_debug_civil_war_probe
			has_country_flag = LSC_debug_war_active
		}
		days_re_enable = 1
		complete_effect = {
			# Step: remove any old mark before applying a new one
			every_other_country = {
				limit = { has_country_flag = LSC_debug_dynamic_rebel }
				log = "[GetDateAsString] LSC_DEBUG IDENTIFY: clearing STALE LSC_debug_dynamic_rebel from [THIS.GetName] ([THIS.GetTag])"
				clr_country_flag = LSC_debug_dynamic_rebel
			}
			# Step: zero the counter
			set_variable = { LSC_debug_rebel_count = 0 }
			# Step: iterate + filter + log every candidate
			every_other_country = {
				limit = {
					original_tag = CZL
					NOT = { tag = CZL }
				}
				log = "[GetDateAsString] LSC_DEBUG IDENTIFY: candidate = [THIS.GetName] ([THIS.GetTag]) government=[THIS.GetRulingIdeology] capital=[THIS.GetCapitalState.GetName] -- cross-check this government/capital against what the test you ran actually expects; has_civil_war status has no confirmed log-interpolation syntax, check the diplomacy screen for that"
				add_to_variable = { LSC_debug_rebel_count = 1 }
			}
			# Step: mark ONLY when exactly 1 candidate was found
			if = {
				limit = {
					check_variable = { LSC_debug_rebel_count = 1 }
				}
				every_other_country = {
					limit = {
						original_tag = CZL
						NOT = { tag = CZL }
					}
					set_country_flag = LSC_debug_dynamic_rebel
					log = "[GetDateAsString] LSC_DEBUG IDENTIFY: MARKED [THIS.GetName] ([THIS.GetTag]) as LSC_debug_dynamic_rebel -- exactly 1 candidate found"
				}
			}
			else = {
				log = "[GetDateAsString] LSC_DEBUG IDENTIFY ERROR: candidate count was NOT exactly 1 (count the 'candidate =' lines above by hand). No flag was set. This is a real blocker for the permanent implementation, not a harness bug — record the exact count."
			}
		}
	}

				limit = { has_country_flag = LSC_debug_dynamic_rebel }
				clr_country_flag = LSC_debug_dynamic_rebel
				log = "[GetDateAsString] LSC_DEBUG CLEAR: removed LSC_debug_dynamic_rebel from [THIS.GetName] ([THIS.GetTag])"
			}
			log = "[GetDateAsString] LSC_DEBUG CLEAR: probe flags/counters cleared on [ROOT.GetName]. No event targets were used anywhere in this harness, so none need clearing. This does NOT end the war, delete the dynamic country, return states, restore units, restore the navy, restore characters, or restore the capital. Reload a clean save before every split test."
		}
