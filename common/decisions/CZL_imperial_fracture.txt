	# Shared availability block for all eight decisions: the crisis must be
	# active, not yet formally closed (LSC_imperial_fracture_closed, set by
	# the three route-choice focuses as a redundant safety layer on top of
	# the route flags themselves), and no heir route may have been chosen
	# yet. This mirrors the category's own `visible` gate (which already
	# hides the whole tab under the same conditions) so a decision can
	# never be reached through this category once the succession is
	# settled, even indirectly.
	#
	# Each decision also blocks itself once its main benefit(s) would be
	# fully wasted by the 0-100 clamp on the crisis variables (Phase 2.5
	# saturation guards), with matching ai_will_do factor = 0 modifiers so
	# the AI never burns political power on a decision that can no longer
	# do anything.
	#
	# AI weights read the six crisis variables directly (no invented "AI
	# leans X" flags) since no route has been chosen yet at this stage.

	LSC_reinforce_alarican_garrisons = {
		icon = GFX_decision_generic_military
		priority = 90

		allowed = {
			original_tag = CZL
		}

		available = {
			has_country_flag = LSC_imperial_fracture_active
			NOT = { has_country_flag = LSC_imperial_fracture_closed }
			NOT = { has_country_flag = LSC_route_kael }
			NOT = { has_country_flag = LSC_route_selene }
			NOT = { has_country_flag = LSC_route_alvaro }
			NOT = {
				check_variable = { LSC_army_loyalty > 94 }
				check_variable = { LSC_revolutionary_strength < 6 }
			}
		}

		cost = 50

		days_re_enable = 90

		complete_effect = {
			add_to_variable = { var = LSC_army_loyalty value = 8 }
			add_to_variable = { var = LSC_revolutionary_strength value = -5 }
			add_to_variable = { var = LSC_northern_unrest value = 10 }
			LSC_clamp_imperial_fracture_variables = yes
			add_manpower = -3000
			custom_effect_tooltip = LSC_reinforce_alarican_garrisons_tt
		}

		ai_will_do = {
			factor = 10
			modifier = { factor = 1.5 check_variable = { LSC_army_loyalty > 55 } }
			modifier = { factor = 1.3 check_variable = { LSC_northern_unrest > 55 } }
			modifier = { factor = 0.5 check_variable = { LSC_revolutionary_strength > 65 } }
			modifier = {
				factor = 0
				check_variable = { LSC_army_loyalty > 94 }
				check_variable = { LSC_revolutionary_strength < 6 }
			}
		}
